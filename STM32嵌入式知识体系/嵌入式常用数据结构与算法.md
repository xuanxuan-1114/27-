# 排序算法
## 冒泡排序
冒泡排序重复比较相邻元素，把较大的元素逐渐移动到数组后面。
```c
void BubbleSort(uint16_t *array, uint16_t length)
{
    if ((array == NULL) || (length < 2U)) {
        return;
    }

    for (uint16_t i = 0U; i < length - 1U; i++) {
        bool swapped = false;

        for (uint16_t j = 0U; j < length - 1U - i; j++) 
        {
            if (array[j] > array[j + 1U]) {
                uint16_t temp = array[j];
                array[j] = array[j + 1U];
                array[j + 1U] = temp;
                swapped = true;
            }
        }

        if (!swapped) {
            break;
        }
    }
```
## 选择排序
每次从未排序区域中选择最小元素，放到前面。
```c
void SelectionSort(uint16_t *array, uint16_t length)
{ 
	if (array == NULL) 
	{ 
		return;
	} 
	for (uint16_t i = 0U; i < length; i++)
	{ 
		uint16_t min_index = i; 
		
		for (uint16_t j = i + 1U; j < length; j++) 
		{ 
			if (array[j] < array[min_index])
			{ 
				min_index = j; 
			} 
		} 
		if (min_index != i) 
		{ 
			uint16_t temp = array[i]; 
			array[i] = array[min_index]; 
			array[min_index] = temp; 
		} 
	} 
}
```
## 插入排序
把每个元素插入前面已经有序的区域
```c
void InsertionSort(uint16_t *array, uint16_t length)
{
    if (array == NULL) {
        return;
    }

    for (uint16_t i = 1U; i < length; i++) {
        uint16_t key = array[i];
        int32_t j = (int32_t)i - 1;

        while ((j >= 0) &&
               (array[j] > key)) {
            array[j + 1] = array[j];
            j--;
        }

        array[j + 1] = key;
    }
}
```
# 环形缓冲区ring buffer
也称作环形队列，是一个用于表示一个固定尺寸，头尾相连的缓冲区的市局结构，适合缓存数据流。
环形缓冲区（ring buffer）特别适合于通信双方循环发送数据的场景；
## 环形缓冲区结构
```c
#define RING_BUFFER_SIZE 128U

typedef struct {
    uint8_t data[RING_BUFFER_SIZE];
    volatile uint16_t read_index;
    volatile uint16_t write_index;
} RingBuffer_t;
```
写入一个字节
```c
bool RingBufferWrite(RingBuffer_t *buffer, uint8_t value)
{
    uint16_t next;

    if (buffer == NULL) {
        return false;
    }

    next = (uint16_t)((buffer->write_index + 1U) %
                      RING_BUFFER_SIZE);

    if (next == buffer->read_index) {
        return false;
    }

    buffer->data[buffer->write_index] = value;
    buffer->write_index = next;

    return true;
}
```
读取一个字节
```c
bool RingBufferRead(RingBuffer_t *buffer, uint8_t *value)
{
    if ((buffer == NULL) || (value == NULL)) {
        return false;
    }

    if (buffer->write_index == buffer->read_index) {
        return false;
    }

    *value = buffer->data[buffer->read_index];

    buffer->read_index =
        (uint16_t)((buffer->read_index + 1U) %
                   RING_BUFFER_SIZE);

    return true;
}
```
判断为空
```c
bool RingBufferIsEmpty(const RingBuffer_t *buffer)
{
    if (buffer == NULL) {
        return true;
    }

    return buffer->write_index == buffer->read_index;
}
```
判断为满
```c
bool RingBufferIsFull(const RingBuffer_t *buffer)
{
    uint16_t next;

    if (buffer == NULL) {
        return false;
    }

    next = (uint16_t)((buffer->write_index + 1U) %
                      RING_BUFFER_SIZE);

    return next == buffer->read_index;
}
```
获取已有数据长度
```c
uint16_t RingBufferLength(const RingBuffer_t *buffer)
{
    if (buffer == NULL) {
        return 0U;
    }

    if (buffer->write_index >= buffer->read_index) {
        return buffer->write_index - buffer->read_index;
    }

    return RING_BUFFER_SIZE -
           buffer->read_index +
           buffer->write_index;
}
```
